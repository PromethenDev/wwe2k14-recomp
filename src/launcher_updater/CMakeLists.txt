cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(wwe2k14_updater LANGUAGES C CXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WWE2K14_UPDATER_SDK "" CACHE PATH "Existing ReXGlue source tree (read only)")
set(WWE2K14_UPDATER_LAUNCHER "" CACHE PATH "Existing launcher2 source tree (read only)")
if(NOT EXISTS "${WWE2K14_UPDATER_SDK}/thirdparty/crypto/sha256.cpp")
  message(FATAL_ERROR "Set WWE2K14_UPDATER_SDK to the ReXGlue source tree")
endif()
if(NOT TARGET wwe2k14_launcher_miniz)
  file(READ "${WWE2K14_UPDATER_LAUNCHER}/thirdparty/miniz/miniz.h" _miniz)
  foreach(_macro MINIZ_NO_INFLATE_APIS MINIZ_NO_ARCHIVE_APIS MINIZ_NO_ARCHIVE_WRITING_APIS
                 MINIZ_NO_ZLIB_APIS MINIZ_SDL_MALLOC MINIZ_STATIC_FUNCTIONS)
    string(REPLACE "#define ${_macro}\n" "" _miniz "${_miniz}")
  endforeach()
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/miniz")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/miniz/miniz.h" "${_miniz}")
  add_library(wwe2k14_launcher_miniz STATIC "${WWE2K14_UPDATER_LAUNCHER}/core/internal/miniz_impl.c")
  target_include_directories(wwe2k14_launcher_miniz PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/miniz")
endif()
add_library(wwe2k14_updater STATIC update.cpp transport.cpp installer.cpp
  "${WWE2K14_UPDATER_SDK}/thirdparty/crypto/sha256.cpp")
target_include_directories(wwe2k14_updater PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE
  "${WWE2K14_UPDATER_SDK}/thirdparty/crypto"
  "${WWE2K14_UPDATER_SDK}/thirdparty/inja/third_party/include")
target_link_libraries(wwe2k14_updater PUBLIC wwe2k14_launcher_miniz)
if(WIN32)
  target_compile_definitions(wwe2k14_updater PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(wwe2k14_updater PRIVATE winhttp user32)
else()
  find_package(CURL 7.85 REQUIRED)
  target_link_libraries(wwe2k14_updater PRIVATE CURL::libcurl)
endif()
if(NOT MSVC)
  target_compile_options(wwe2k14_updater PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  enable_testing()
  add_executable(updater-test test.cpp)
  target_link_libraries(updater-test PRIVATE wwe2k14_updater)
  target_include_directories(updater-test PRIVATE "${WWE2K14_UPDATER_SDK}/thirdparty/inja/third_party/include")
  add_test(NAME updater-test COMMAND updater-test)
endif()
